1. Get the status of the deployment:
  kubectl get deployment -n {{ .Values.global.namespace }}

2. Forward the metrics port to your local machine:
  kubectl port-forward -n {{ .Values.global.namespace }} svc/{{ include "advanced-sftp-exporter.fullname" . }} 9115:9115

3. Access Prometheus metrics:
  http://localhost:9115/metrics

4. Check the health of the exporter:
  kubectl exec -it -n {{ .Values.global.namespace }} deployments/{{ include "advanced-sftp-exporter.fullname" . }} -- \
    wget -q -O - http://localhost:9115/health

5. View the logs:
  kubectl logs -n {{ .Values.global.namespace }} -f deployment/{{ include "advanced-sftp-exporter.fullname" . }}

6. Configure Prometheus to scrape the exporter:
  Add the following to your prometheus.yml:
  
  scrape_configs:
    - job_name: 'sftp-exporter'
      static_configs:
        - targets: ['{{ include "advanced-sftp-exporter.fullname" . }}.{{ .Values.global.namespace }}.svc.cluster.local:9115']

{{- if .Values.prometheus.serviceMonitor.enabled }}

7. ServiceMonitor is enabled. Configure Prometheus to use it:
  prometheus:
    prometheusSpec:
      serviceMonitorSelectorNilUsesHelmValues: false

{{- end }}
